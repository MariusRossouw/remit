"A wallet for sending and requesting money."
interface Wallet {
  "Read the current balance."
  action balance() -> Balance
    effects: read

  "Pay money to another party. Takes at most 1000 ZAR per call and does nothing else. Cannot be undone."
  action transfer(to: Text, amount: Money, note: Text?) -> Transfer
    effects: spend(max: 1000 ZAR)

  "Ask a payer for money. Contacts the payer on the caller's behalf and changes nothing else. Can be undone with withdrawRequest."
  action requestMoney(payer: Text, amount: Money) -> MoneyRequest
    effects: send
    undo: withdrawRequest

  "Withdraw a money request. Contacts the payer on the caller's behalf and changes nothing else. Cannot be undone."
  action withdrawRequest(request: MoneyRequest)
    effects: send
}

"The money held in a wallet."
type Balance { amount: Money }

"A payment made from a wallet."
type Transfer { id: Text  to: Text  amount: Money }

"A request for money sent to a payer."
type MoneyRequest { id: Text  payer: Text  amount: Money }
