```yaml
openapi: 3.1.0
info:
  title: Home
  version: 1.0.0
  description: Controls devices in a house.
paths:
  /devices:
    get:
      operationId: listDevices
      summary: List devices
      description: Returns all devices in the house. Read-only; changes no data.
      x-changes-data: false
      responses:
        "200":
          description: The devices in the house.
          content:
            application/json:
              schema:
                type: array
                items:
                  $ref: "#/components/schemas/Device"
  /devices/{deviceId}/temperature:
    post:
      operationId: setTemperature
      summary: Set temperature
      description: >-
        Sets the target temperature of a thermostat and returns the resulting Setting.
        Changes data. This change can be undone with restore, for one hour afterwards.
      x-changes-data: true
      x-undoable: true
      x-undo-window: PT1H
      x-undo-operation: restore
      parameters:
        - name: deviceId
          in: path
          required: true
          description: Identifier of the thermostat.
          schema:
            type: string
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required:
                - celsius
              properties:
                celsius:
                  type: number
                  description: Target temperature in degrees Celsius.
              additionalProperties: false
      responses:
        "200":
          description: The change made to the device's setting.
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Setting"
  /restore:
    post:
      operationId: restore
      summary: Restore setting
      description: >-
        Restores a device to the previous value recorded in the given Setting.
        Changes data. This change cannot be undone.
      x-changes-data: true
      x-undoable: false
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required:
                - setting
              properties:
                setting:
                  $ref: "#/components/schemas/Setting"
              additionalProperties: false
      responses:
        "204":
          description: The setting was restored. No content is returned.
  /devices/{deviceId}/unlock:
    post:
      operationId: unlock
      summary: Unlock device
      description: >-
        Unlocks a lock for the given number of minutes. Changes data. This change cannot be undone.
      x-changes-data: true
      x-undoable: false
      parameters:
        - name: deviceId
          in: path
          required: true
          description: Identifier of the lock.
          schema:
            type: string
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required:
                - minutes
              properties:
                minutes:
                  type: integer
                  description: Number of minutes to keep the lock unlocked.
              additionalProperties: false
      responses:
        "204":
          description: The lock was unlocked. No content is returned.
components:
  schemas:
    DeviceKind:
      type: string
      enum:
        - light
        - lock
        - thermostat
    Device:
      type: object
      required:
        - id
        - name
        - kind
      properties:
        id:
          type: string
        name:
          type: string
        kind:
          $ref: "#/components/schemas/DeviceKind"
      additionalProperties: false
    Setting:
      type: object
      required:
        - deviceId
        - previous
        - current
      properties:
        deviceId:
          type: string
        previous:
          type: number
          description: Value before the change.
        current:
          type: number
          description: Value after the change.
      additionalProperties: false
```
