"A clinic where patients book appointments."

type Slot { id: Text  start: DateTime  doctor: Text }

type Appointment { id: Text  slot: Slot  patient: Text }

"A clinic where patients book appointments."
interface Clinic {
  "Find the open slots on a day."
  action findSlots(day: Date) -> List<Slot>
    effects: read

  "Book a slot for a patient. Can be undone with cancel."
  action book(slot: Slot, patient: Text) -> Appointment
    effects: write
    undo: cancel

  "Cancel an appointment. This cannot be undone."
  action cancel(appointment: Appointment)
    effects: write
}
