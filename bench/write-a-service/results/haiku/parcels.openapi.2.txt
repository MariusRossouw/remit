```yaml
openapi: 3.1.0
info:
  title: Parcels
  version: 1.0.0
  summary: A courier that quotes for, ships, and tracks parcels.
  description: >-
    Parcels is a courier service. Clients request a quote for a parcel, ship the
    parcel against that quote, and track the shipment afterwards.
paths:
  /quotes:
    get:
      operationId: quote
      summary: Quote the price and transit time for a parcel.
      description: >-
        Read-only. Returns a Quote for moving a parcel of the given weight from
        origin to destination. Does not change data and does not take payment.
      x-read-only: true
      parameters:
        - name: origin
          in: query
          required: true
          description: Origin address.
          schema:
            type: string
        - name: destination
          in: query
          required: true
          description: Destination address.
          schema:
            type: string
        - name: grams
          in: query
          required: true
          description: Parcel weight in grams.
          schema:
            type: integer
            minimum: 1
      responses:
        "200":
          description: The quote for the parcel.
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Quote"
        "400":
          description: A parameter is missing or invalid.
        "404":
          description: The origin or destination is not served.
  /shipments:
    post:
      operationId: ship
      summary: Ship a parcel against a quote.
      description: >-
        Changes data and takes payment for the quoted price. This action cannot
        be undone. Clients must not retry automatically without first checking
        whether a shipment was already created.
      x-irreversible: true
      x-takes-payment: true
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              additionalProperties: false
              required:
                - quote
                - to
              properties:
                quote:
                  $ref: "#/components/schemas/Quote"
                to:
                  type: string
                  description: Recipient address.
      responses:
        "201":
          description: The shipment was created and payment was taken.
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Shipment"
        "400":
          description: The request is invalid.
        "402":
          description: Payment was declined. No shipment was created.
        "409":
          description: The quote has expired or was already used.
  /shipments/{shipmentId}/tracking:
    get:
      operationId: track
      summary: List tracking events for a shipment.
      description: >-
        Read-only. Returns the tracking events for a shipment, in the order the
        service received them. Does not change data.
      x-read-only: true
      parameters:
        - name: shipmentId
          in: path
          required: true
          description: The identifier of the shipment.
          schema:
            type: string
      responses:
        "200":
          description: The tracking events for the shipment.
          content:
            application/json:
              schema:
                type: array
                items:
                  $ref: "#/components/schemas/TrackingEvent"
        "404":
          description: No shipment has this identifier.
components:
  schemas:
    Money:
      type: object
      required:
        - amount
        - currency
      additionalProperties: false
      properties:
        amount:
          type: string
          description: Decimal amount as a string, to avoid floating-point rounding.
          pattern: "^[0-9]+(\\.[0-9]{1,2})?$"
        currency:
          type: string
          description: ISO 4217 currency code.
          pattern: "^[A-Z]{3}$"
    Quote:
      type: object
      required:
        - id
        - price
        - days
      additionalProperties: false
      properties:
        id:
          type: string
        price:
          $ref: "#/components/schemas/Money"
        days:
          type: integer
          description: Expected transit time in whole days.
          minimum: 0
    Shipment:
      type: object
      required:
        - id
        - quote
        - to
      additionalProperties: false
      properties:
        id:
          type: string
        quote:
          $ref: "#/components/schemas/Quote"
        to:
          type: string
          description: Recipient address.
    TrackingEvent:
      type: object
      required:
        - at
        - note
      additionalProperties: false
      properties:
        at:
          type: string
          format: date-time
          description: When the event happened.
        note:
          type: string
          description: Text written by couriers, not by this service.
```
