info:

      summary: List events between two dates

      x-contacts-attendees: false
      parameters:
        - name: first
          in: query
          required: true
          schema:
            type: string
            format: date
        - name: last
          in: query
          required: true
          schema:
            type: string
            format: date
      responses:
        "200":
          description: A list of events.
          content:
            application/json:
              schema:
                type: array
                items:
                  $ref: "#/components/schemas/Event"
    post:
      operationId: createEvent
      summary: Create an event
      description: >-
        Changes data and contacts the attendees on the caller's behalf.
        Can be undone with deleteEvent.
      x-reads-only: false
      x-changes-data: true
      x-contacts-attendees: true
      x-reversible: true

      requestBody:
        required: true

          application/json:
            schema:
              type: object
              required: [title, start, end, attendees]
              additionalProperties: false
              properties:
                title:
                  type: string
                start:
                  type: string
                  format: date-time
                end:
                  type: string
                  format: date-time
                attendees:
                  type: array
                  items:
                    type: string
      responses:
        "201":
          description: The created event.
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Event"

    delete:
      operationId: deleteEvent
      summary: Delete an event
      description: >-
        Changes data and contacts the attendees on the caller's behalf.

      x-reads-only: false
      x-changes-data: true
      x-contacts-attendees: true
      x-reversible: false
      parameters:
        - name: event
          in: path
          required: true
          description: The id of the Event to delete.
          schema:
            type: string
      responses:
        "204":
          description: Deleted. Returns nothing.

          type: string
          format: date-time
        end:

          format: date-time
        attendees:
