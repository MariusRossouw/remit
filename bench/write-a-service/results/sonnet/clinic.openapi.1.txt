```yaml
openapi: 3.1.0
info:
  title: Clinic
  version: 1.0.0
  description: A clinic where patients book appointments.
paths:
  /slots:
    get:
      operationId: findSlots
      summary: Find slots on a given day
      description: Only reads data. Safe and idempotent.
      x-side-effects: none
      parameters:
        - name: day
          in: query
          required: true
          schema:
            type: string
            format: date
      responses:
        "200":
          description: A list of slots.
          content:
            application/json:
              schema:
                type: array
                items:
                  $ref: "#/components/schemas/Slot"
  /appointments:
    post:
      operationId: book
      summary: Book an appointment
      description: Changes data. Can be undone with cancel.
      x-side-effects: write
      x-reversible: true
      x-undone-by: cancel
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required: [slot, patient]
              properties:
                slot:
                  $ref: "#/components/schemas/Slot"
                patient:
                  type: string
      responses:
        "200":
          description: The created appointment.
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Appointment"
          links:
            undo:
              operationId: cancel
              description: Undo this booking by cancelling the appointment.
              requestBody:
                appointment: "$response.body"
  /appointments/cancel:
    post:
      operationId: cancel
      summary: Cancel an appointment
      description: Changes data. Cannot be undone.
      x-side-effects: write
      x-reversible: false
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required: [appointment]
              properties:
                appointment:
                  $ref: "#/components/schemas/Appointment"
      responses:
        "204":
          description: Appointment cancelled. No content returned.
components:
  schemas:
    Slot:
      type: object
      required: [id, start, doctor]
      properties:
        id:
          type: string
        start:
          type: string
          format: date-time
        doctor:
          type: string
    Appointment:
      type: object
      required: [id, slot, patient]
      properties:
        id:
          type: string
        slot:
          $ref: "#/components/schemas/Slot"
        patient:
          type: string
```
